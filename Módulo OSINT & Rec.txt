5. Rota de API no Backend Next.js (src/app/api/osint/investigate/route.ts)
TypeScript
import { NextResponse } from 'next/server';
import { PrismaClient } from '@prisma/client';

const prisma = new PrismaClient();

export async function POST(req: Request) {
  try {
    const { targetType, value } = await req.json();

    if (!targetType || !value) {
      return NextResponse.json({ error: 'Parâmetros inválidos.' }, { status: 400 });
    }

    const startTime = Date.now();

    // 1. Registra a busca como PENDING no banco de dados
    const scanRecord = await prisma.osintScan.create({
      data: {
        targetType,
        targetValue: value,
        status: 'PROCESSING',
      },
    });

    // 2. Faz a chamada ao container Python dedicado
    const response = await fetch('http://osint-brazuca-engine:8001/search', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'X-API-Token': process.env.OSINT_ENGINE_SECRET || 'mcinfrati_osint_secret_2026',
      },
      body: JSON.stringify({ target_type: targetType, value }),
    });

    if (!response.ok) {
      throw new Error('Falha ao obter resposta da engine de OSINT.');
    }

    const resultData = await response.json();
    const executionTime = Date.now() - startTime;

    // 3. Atualiza o registro com os resultados retornados
    const updatedRecord = await prisma.osintScan.update({
      where: { id: scanRecord.id },
      data: {
        status: 'COMPLETED',
        rawResult: resultData,
        executionTime,
      },
    });

    return NextResponse.json(updatedRecord);
  } catch (error: any) {
    return NextResponse.json({ error: error.message }, { status: 500 });
  }
}

'use client';

import React, { useState } from 'react';
import { Search, ShieldAlert, Loader2, Database, Globe, UserCheck, FileText } from 'lucide-react';

export const OsintReconView = () => {
  const [targetType, setTargetType] = useState('CNPJ');
  const [targetValue, setTargetValue] = useState('');
  const [loading, setLoading] = useState(false);
  const [results, setResults] = useState<any>(null);

  const handleSearch = async () => {
    if (!targetValue) return;
    setLoading(true);
    try {
      const response = await fetch('/api/osint/investigate', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ targetType, value: targetValue }),
      });
      const data = await response.json();
      setResults(data);
    } catch (err) {
      console.error(err);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="p-6 max-w-7xl mx-auto space-y-6 text-slate-100">
      <div className="border-b border-slate-800 pb-4">
        <h1 className="text-2xl font-bold flex items-center gap-2 text-white">
          <Search className="w-7 h-7 text-blue-500" />
          OSINT & Recon (Brazuca Engine)
        </h1>
        <p className="text-sm text-slate-400 mt-1">
          Investigação de fontes abertas sobre alvos brasileiros (CPF, CNPJ, Domínios, E-mails e Telefones).
        </p>
      </div>

      {/* BARRA DE PESQUISA */}
      <div className="bg-slate-900 border border-slate-800 p-4 rounded-xl flex gap-3">
        <select
          value={targetType}
          onChange={(e) => setTargetType(e.target.value)}
          className="bg-slate-950 border border-slate-800 rounded-lg px-3 py-2 text-sm text-slate-200 focus:outline-none focus:border-blue-500"
        >
          <option value="CNPJ">CNPJ</option>
          <option value="CPF">CPF</option>
          <option value="EMAIL">E-mail</option>
          <option value="PHONE">Telefone</option>
          <option value="DOMAIN">Domínio / IP</option>
          <option value="NAME">Nome Completo</option>
        </select>

        <input
          type="text"
          placeholder={`Digite o ${targetType}...`}
          value={targetValue}
          onChange={(e) => setTargetValue(e.target.value)}
          className="flex-1 bg-slate-950 border border-slate-800 rounded-lg px-4 py-2 text-sm text-slate-200 focus:outline-none focus:border-blue-500"

        <button

          className="bg-blue-600 hover:bg-blue-500 disabled:opacity-50 text-white px-5 py-2 rounded-lg text-sm font-medium transition-colors flex items-center gap-2"

            <h3 className="text-sm font-semibold flex items-center gap-2 text-blue-400">
