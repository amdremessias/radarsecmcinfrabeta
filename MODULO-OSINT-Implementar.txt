🌐 4. API ROUTE NEXT.JS COM TIMEOUT E FALLBACK (src/app/api/osint/investigate/route.ts)
TypeScript
import { NextResponse } from 'next/server';
import { PrismaClient } from '@prisma/client';

const prisma = new PrismaClient();

export async function POST(req: Request) {
  try {
    const { targetType, value } = await req.json();

    if (!targetType || !value) {
      return NextResponse.json({ error: 'Parâmetros inválidos.' }, { status: 400 });
    }

    const startTime = Date.now();

    // 1. Cria o registro PENDING
    const scanRecord = await prisma.osintScan.create({
      data: {
        targetType,
        targetValue: value,
        status: 'PROCESSING',
      },
    });

    // 2. Chamada com AbortController para Timeout de 15 segundos
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 15000);

    const engineUrl = process.env.OSINT_ENGINE_URL || 'http://osint-brazuca-engine:8001';
    const secret = process.env.OSINT_ENGINE_SECRET || 'mcinfrati_osint_secret_2026';

    let resultData = null;
    let scanStatus: 'COMPLETED' | 'FAILED' = 'COMPLETED';

    try {
      const response = await fetch(`${engineUrl}/search`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-API-Token': secret,
        },
        body: JSON.stringify({ target_type: targetType, value }),
        signal: controller.signal,
      });

      clearTimeout(timeoutId);

      if (!response.ok) throw new Error('Falha na resposta do container OSINT.');
      resultData = await response.json();
    } catch (err: any) {
      scanStatus = 'FAILED';
      resultData = { error: err.name === 'AbortError' ? 'Timeout de consulta excedido (15s).' : err.message };
    }

    const executionTime = Date.now() - startTime;

    // 3. Atualiza o registro no banco
    const updatedRecord = await prisma.osintScan.update({
      where: { id: scanRecord.id },
      data: {
        status: scanStatus,
        rawResult: resultData,
        executionTime,
      },
    });

    return NextResponse.json(updatedRecord, { status: scanStatus === 'FAILED' ? 502 : 200 });
  } catch (error: any) {
    return NextResponse.json({ error: error.message }, { status: 500 });
  }
}

'use client';

import React, { useState } from 'react';
import { ShieldAlert, Search, Loader2, Database, Globe, UserCheck, Check, Copy } from 'lucide-react';

export default function SecurityAdminPanel() {
  const [activeTab, setActiveTab] = useState<'sast' | 'osint'>('sast');
  const [targetType, setTargetType] = useState('CNPJ');
  const [targetValue, setTargetValue] = useState('');
  const [osintLoading, setOsintLoading] = useState(false);
  const [osintResult, setOsintResult] = useState<any>(null);
  const [copied, setCopied] = useState(false);

  const handleOsintSearch = async () => {
    if (!targetValue) return;
    setOsintLoading(true);
    try {
      const res = await fetch('/api/osint/investigate', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ targetType, value: targetValue }),
      });
      const data = await res.json();
      setOsintResult(data);
    } catch (e) {
      console.error(e);
    } finally {
      setOsintLoading(false);
    }
  };

  const copyAsMarkdown = () => {
    if (!osintResult) return;
    const md = `### [OSINT] Investigação de Alvo (${osintResult.targetType})
**Valor:** \`${osintResult.targetValue}\`
**Status:** \`${osintResult.status}\`

\`\`\`json
${JSON.stringify(osintResult.rawResult, null, 2)}
\`\`\``;
    navigator.clipboard.writeText(md);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  };

  return (
    <div className="min-h-screen bg-slate-950 text-slate-100 p-8 space-y-6">
      <div className="flex justify-between items-center border-b border-slate-800 pb-4">
        <h1 className="text-2xl font-bold text-white flex items-center gap-2">
          <ShieldAlert className="w-7 h-7 text-blue-500"/>
          MCinfraTI Radar Security
        </h1>
        <div className="flex gap-2 bg-slate-900 p-1 rounded-lg border border-slate-800">
          <button
            onClick={() => setActiveTab('sast')}
            className={`px-4 py-2 text-sm font-medium rounded-md transition-colors ${
              activeTab === 'sast' ? 'bg-blue-600 text-white' : 'text-slate-400 hover:text-white'
            }`}
          >
            Análise Profunda (SAST)
          </button>
          <button
            onClick={() => setActiveTab('osint')}
            className={`px-4 py-2 text-sm font-medium rounded-md transition-colors ${
              activeTab === 'osint' ? 'bg-blue-600 text-white' : 'text-slate-400 hover:text-white'
            }`}
          >
            OSINT & Recon
          </button>
        </div>
      </div>

      {activeTab === 'sast' && (
        <div className="bg-slate-900 border border-slate-800 rounded-xl p-6 space-y-4">
          <h2 className="text-lg font-semibold text-slate-200">Submeter Código para Auditoria Estática</h2>
          <div className="grid grid-cols-2 gap-4">
            <input
              type="text"
              placeholder="URL do Repositório Git"
              className="bg-slate-950 border border-slate-800 rounded-lg px-4 py-2 text-sm focus:outline-none focus:border-blue-500"
            />
            <input
              type="text"
              placeholder="Branch (default: main)"
              className="bg-slate-950 border border-slate-800 rounded-lg px-4 py-2 text-sm focus:outline-none focus:border-blue-500"
            />
          </div>
          <button className="bg-blue-600 hover:bg-blue-500 text-white px-5 py-2 rounded-lg text-sm font-medium transition-colors">
            Iniciar Varredura SAST

    expect(res.body).toHaveProperty('id');
    expect(res.body.status).toBe('COMPLETED');
