Componente do Botão de Exportação (src/components/ExportPdfButton.tsx)
TypeScript
'use client';

import React, { useState } from 'react';
import { Download, Loader2, CheckCircle, AlertCircle } from 'lucide-react';

interface ExportPdfButtonProps {
  scanId: string;
  targetUrl?: string;
  className?: string;
}

export const ExportPdfButton: React.FC<ExportPdfButtonProps> = ({
  scanId,
  targetUrl = 'relatorio',
  className = '',
}) => {
  const [loading, setLoading] = useState(false);
  const [status, setStatus] = useState<'idle' | 'success' | 'error'>('idle');

  const handleDownloadPdf = async () => {
    try {
      setLoading(true);
      setStatus('idle');

      // Chamada para a API do backend que gera o PDF via Puppeteer
      const response = await fetch(`/api/scans/${scanId}/report/pdf`, {
        method: 'GET',
        headers: {
          'Accept': 'application/pdf',
        },
      });

      if (!response.ok) {
        throw new Error('Falha ao gerar o relatório em PDF.');
      }

      // Converte a resposta em Blob e dispara o download no navegador
      const blob = await response.blob();
      const url = window.URL.createObjectURL(blob);
      const link = document.createElement('a');
      
      // Sanitiza o nome do arquivo
      const sanitizedUrl = targetUrl.replace(/[^a-zA-Z0-9]/g, '_');
      link.href = url;
      link.download = `MCinfraTI_Radar_${sanitizedUrl}_${scanId.slice(0, 8)}.pdf`;
      
      document.body.appendChild(link);
      link.click();
      
      // Limpeza de memória
      document.body.removeChild(link);
      window.URL.revokeObjectURL(url);

      setStatus('success');
      setTimeout(() => setStatus('idle'), 3000);
    } catch (error) {
      console.error('[ExportPdf] Erro ao baixar o PDF:', error);
      setStatus('error');
      setTimeout(() => setStatus('idle'), 4000);
    } finally {
      setLoading(false);
    }
  };

  return (
    <button
      onClick={handleDownloadPdf}
      disabled={loading}
      className={`inline-flex items-center gap-2 px-4 py-2 text-sm font-medium rounded-lg transition-all duration-200 border shadow-sm select-none
        ${
          status === 'error'
            ? 'bg-red-500/10 border-red-500/30 text-red-400'
            : status === 'success'
            ? 'bg-emerald-500/10 border-emerald-500/30 text-emerald-400'
            : 'bg-slate-800 hover:bg-slate-700 border-slate-700 hover:border-slate-600 text-slate-200 hover:text-white'
        }
        disabled:opacity-50 disabled:cursor-not-allowed ${className}`}
    >
      {loading ? (
        <>
          <Loader2 className="w-4 h-4 animate-spin text-blue-400" />
          <span>Gerando PDF...</span>
        </>
      ) : status === 'success' ? (
        <>
          <CheckCircle className="w-4 h-4 text-emerald-400" />
          <span>Download Concluído!</span>
        </>
      ) : status === 'error' ? (
        <>
          <AlertCircle className="w-4 h-4 text-red-400" />
          <span>Erro ao Exportar</span>
        </>
      ) : (
        <>
          <Download className="w-4 h-4 text-blue-400" />
          <span>Exportar PDF</span>
        </>
      )}
    </button>
  );
};
